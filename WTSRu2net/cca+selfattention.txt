# -*- coding: utf-8 -*-

import torch
import torch.nn as nn
import torch.nn.functional as F
import einops

# 导入 U2Net 基础模块
from u2net import REBNCONV, RSU7, RSU6, RSU5, RSU4, RSU4F
# 导入 DCA 组件 (解耦使用)
from model.utils.dca import ChannelAttention, SpatialAttention
# 导入 RRDB 超分辨率模块
from rrdb import RRDBNet


##############################################################################
# 辅助函数：下采样 -> RRDB 超分 -> [新增] SCA 空间增强 -> 尺寸匹配
##############################################################################
def _rrdb_upsample_like(src, tar, rrdb_net, sca_module=None):
    """
    辅助函数：下采样 -> RRDB 超分 -> 尺寸匹配
    """
    # 模拟退化
    down = F.interpolate(src, scale_factor=0.5, mode='bicubic', align_corners=False)
    # RRDB 超分重建
    sr = rrdb_net(down)

    # [新增] 引入 SCA 增强空间边缘感知
    if sca_module is not None:
        b, c, h, w = sr.shape
        # 将特征图展平为 Tokens
        x_token = einops.rearrange(sr, 'b c h w -> b (h w) c')
        # 利用全局空间布局纠正局部伪影
        refined_token = sca_module([x_token, x_token, x_token])
        sr = einops.rearrange(refined_token, 'b (h w) c -> b c h w', h=h, w=w)

    # 匹配目标尺寸
    sr = F.interpolate(sr, size=tar.shape[2:], mode='bicubic', align_corners=False)
    return sr


class U2NET_RRDB(nn.Module):
    def __init__(self, in_ch=3, out_ch=1):
        super(U2NET_RRDB, self).__init__()

        # ---- Encoder (编码器阶段) ----
        self.stage1 = RSU7(in_ch, 32, 128)
        self.pool12 = nn.MaxPool2d(2, stride=2, ceil_mode=True)
        self.stage2 = RSU6(128, 32, 128)
        self.pool23 = nn.MaxPool2d(2, stride=2, ceil_mode=True)
        self.stage3 = RSU5(128, 32, 128)
        self.pool34 = nn.MaxPool2d(2, stride=2, ceil_mode=True)
        self.stage4 = RSU4(128, 32, 128)
        self.pool45 = nn.MaxPool2d(2, stride=2, ceil_mode=True)
        self.stage5 = RSU4F(128, 32, 128)
        self.pool56 = nn.MaxPool2d(2, stride=2, ceil_mode=True)
        self.stage6 = RSU4F(128, 32, 128)

        # ---- [核心修改] 解耦 DCA 组件初始化 ----

        # 1. CCA Bridge: 专门用于 5 个跳跃连接的通道语义对齐
        self.cca_stages = nn.ModuleList([
            ChannelAttention(in_features=128 * 5, out_features=128, n_heads=1)
            for _ in range(5)
        ])

        # 2. SCA Gate: 专门用于超分模块的空间结构引导
        self.sca_gate = SpatialAttention(in_features=128, out_features=128, n_heads=1)

        # 针对 256 分辨率的最优 Token 尺寸配置
        self.patch_embed = nn.AdaptiveAvgPool2d((16, 16))
        self.cca_norm = nn.LayerNorm(128)

        # ---- Decoder (解码器阶段) ----
        self.stage5d = RSU4F(256, 64, 128)
        self.stage4d = RSU4(256, 64, 128)
        self.stage3d = RSU5(256, 64, 128)
        self.stage2d = RSU6(256, 64, 128)
        self.stage1d = RSU7(256, 64, 128)

        # Side Outputs (侧边输出)
        self.side1 = nn.Conv2d(128, out_ch, 3, padding=1)
        self.side2 = nn.Conv2d(128, out_ch, 3, padding=1)
        self.side3 = nn.Conv2d(128, out_ch, 3, padding=1)
        self.side4 = nn.Conv2d(128, out_ch, 3, padding=1)
        self.side5 = nn.Conv2d(128, out_ch, 3, padding=1)
        self.side6 = nn.Conv2d(128, out_ch, 3, padding=1)

        self.outconv = nn.Conv2d(6 * out_ch, out_ch, 1)

        # RRDB 超分上采样模块
        self.rrdb_up = RRDBNet(in_nc=128, out_nc=128, nf=128, nb=3, growth_channels=32, scale=2)

        self.epoch = None

    def set_epoch(self, epoch):
        self.epoch = epoch
        if hasattr(self, 'rrdb_up'):
            self.rrdb_up.set_epoch(epoch)

    def forward(self, x):
        # ---- Encoder Forward ----
        hx1 = self.stage1(x)
        hx2 = self.stage2(self.pool12(hx1))
        hx3 = self.stage3(self.pool23(hx2))
        hx4 = self.stage4(self.pool34(hx3))
        hx5 = self.stage5(self.pool45(hx4))
        hx6 = self.stage6(self.pool56(hx5))

        # ---- [修改点 1] CCA 增强跳跃连接 (解耦版) ----
        raw_feats = [hx1, hx2, hx3, hx4, hx5]
        # 统一尺寸池化并平铺
        tokens = [einops.rearrange(self.patch_embed(f), 'b c h w -> b (h w) c') for f in raw_feats]
        # 跨尺度通道拼接
        all_tokens = torch.cat(tokens, dim=2)

        en_feats = []
        for i, q_token in enumerate(tokens):
            q_token = self.cca_norm(q_token)
            # 应用通道注意力校准
            att_token = self.cca_stages[i]([q_token, all_tokens, all_tokens])
            # 还原形状并残差融合
            f_en = einops.rearrange(att_token, 'b (h w) c -> b c h w', h=16, w=16)
            f_en = F.interpolate(f_en, size=raw_feats[i].shape[2:], mode='bilinear', align_corners=False)
            en_feats.append(f_en + raw_feats[i])

        hx1_en, hx2_en, hx3_en, hx4_en, hx5_en = en_feats

        # ---- [修改点 2] Decoder Forward (SCA 引导的上采样) ----
        # 调用时传入 sca_gate 以实现空间增强
        hx6up = _rrdb_upsample_like(hx6, hx5_en, self.rrdb_up, self.sca_gate)
        hx5d = self.stage5d(torch.cat((hx6up, hx5_en), 1))

        hx5dup = _rrdb_upsample_like(hx5d, hx4_en, self.rrdb_up, self.sca_gate)
        hx4d = self.stage4d(torch.cat((hx5dup, hx4_en), 1))

        hx4dup = _rrdb_upsample_like(hx4d, hx3_en, self.rrdb_up, self.sca_gate)
        hx3d = self.stage3d(torch.cat((hx4dup, hx3_en), 1))

        hx3dup = _rrdb_upsample_like(hx3d, hx2_en, self.rrdb_up, self.sca_gate)
        hx2d = self.stage2d(torch.cat((hx3dup, hx2_en), 1))

        hx2dup = _rrdb_upsample_like(hx2d, hx1_en, self.rrdb_up, self.sca_gate)
        hx1d = self.stage1d(torch.cat((hx2dup, hx1_en), 1))

        # ---- Side Outputs & Final Output ----
        d1 = self.side1(hx1d)
        d2 = self.side2(hx2d)
        d3 = self.side3(hx3d)
        d4 = self.side4(hx4d)
        d5 = self.side5(hx5d)
        d6 = self.side6(hx6)

        d2 = F.interpolate(d2, size=d1.shape[2:], mode='bilinear', align_corners=False)
        d3 = F.interpolate(d3, size=d1.shape[2:], mode='bilinear', align_corners=False)
        d4 = F.interpolate(d4, size=d1.shape[2:], mode='bilinear', align_corners=False)
        d5 = F.interpolate(d5, size=d1.shape[2:], mode='bilinear', align_corners=False)
        d6 = F.interpolate(d6, size=d1.shape[2:], mode='bilinear', align_corners=False)

        d0 = self.outconv(torch.cat((d1, d2, d3, d4, d5, d6), dim=1))

        return (torch.sigmoid(d0), torch.sigmoid(d1), torch.sigmoid(d2),
                torch.sigmoid(d3), torch.sigmoid(d4), torch.sigmoid(d5), torch.sigmoid(d6))
